cmake_minimum_required(VERSION 3.16)
project(lattice_ros2_gateway LANGUAGES CXX)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 20)
endif()

set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(nlohmann_json REQUIRED)

add_executable(lattice_ros2_gateway src/main.cpp)
target_compile_options(lattice_ros2_gateway PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(lattice_ros2_gateway PRIVATE include)
target_link_libraries(lattice_ros2_gateway nlohmann_json::nlohmann_json)
ament_target_dependencies(lattice_ros2_gateway rclcpp rclcpp_lifecycle rclcpp_action lifecycle_msgs nav2_msgs geometry_msgs std_msgs sensor_msgs tf2_ros)

if(BUILD_TESTING)
  enable_testing()
  add_executable(motion_gate_test test/motion_gate_test.cpp)
  target_include_directories(motion_gate_test PRIVATE include)
  target_compile_options(motion_gate_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME motion_gate COMMAND motion_gate_test)
  add_executable(route_geometry_test test/route_geometry_test.cpp)
  target_include_directories(route_geometry_test PRIVATE include)
  target_compile_options(route_geometry_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME route_geometry COMMAND route_geometry_test)
endif()

install(TARGETS lattice_ros2_gateway DESTINATION lib/${PROJECT_NAME})
ament_package()
