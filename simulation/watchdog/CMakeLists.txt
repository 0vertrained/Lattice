cmake_minimum_required(VERSION 3.16)
project(lattice_sim_watchdog LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(lattice-sim-watchdog SHARED watchdog.cpp)
target_link_libraries(lattice-sim-watchdog PRIVATE gz-sim8::gz-sim8 gz-plugin2::register)
target_compile_options(lattice-sim-watchdog PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS lattice-sim-watchdog LIBRARY DESTINATION lib)
